## NVSwitch 候选 GPU 剪枝

新版由 ncclTopoSearchNextGpuSortNvl 直接生成邻居候选，不再先排序再搬移 next 数组。

原文件：`src/graph/search.cc`，以下为原有注释及邻近代码的摘录。

```cpp

  *countPtr = count;  // 先写入全量个数

  if (system->nodes[NVS].count) {  // 全互联，整条环被锁死成 PCI 连续游走，排列空间从 (ngpus-1)! 塌缩到 2（升序与降序）
    // NVSwitches prefer when we talk to a limited set of peers. Try to use neighbors first.
    int index = gpu - system->nodes[GPU].nodes;
    int i;
    // 计算 PCI 左右邻居 GPU
    int prevGpu = (index - 1 + ngpus) % ngpus;
    int nextGpu = (index + 1) % ngpus;
    int firstGpus[2];
    int firstGpuCount = 0;
    /*
    按 pattern 决定优先邻居：
      pattern	                                        firstGpus 数组	        个数
      RING	                                          {nextGpu, prevGpu}	    2
      SPLIT_TREE / BALANCED_TREE	                    {prevGpu, nextGpu}	    2
      TREE / NVLS / COLLNET_DIRECT / COLLNET_CHAIN	  {nextGpu}              	1
    */
    if (graph->pattern == NCCL_TOPO_PATTERN_RING) {
      firstGpus[0] = nextGpu;
      firstGpus[1] = prevGpu;
      firstGpuCount = 2;
    } else if (graph->pattern == NCCL_TOPO_PATTERN_SPLIT_TREE || graph->pattern == NCCL_TOPO_PATTERN_BALANCED_TREE) {
      firstGpus[0] = prevGpu;
      firstGpus[1] = nextGpu;
      firstGpuCount = 2;
    } else {
      firstGpus[0] = nextGpu;
      firstGpuCount = 1;
    }
    // ngpus == 2 时 (index+1)%2 与 (index-1+2)%2 都等于对方，两个候选重复
    if (nextGpu == prevGpu && firstGpuCount == 2) firstGpuCount = 1;
    int firstGpuRealCount = 0;  // 记录实际成功提升的邻居数，最终成为 *countPtr。取值只能是 0、1、2
    for (int g = 0; g < firstGpuCount; g++) {
      for (i = 0; i < count && next[i] != firstGpus[g]; i++);
      if (i < count) {  // 找到候选 GPU 中的 firstGpus[g]
        for (; i > 0; i--) next[i] = next[i - 1];  // 把 [0, i-1] 整体后移到 [1, i]
        next[0] = firstGpus[g];  // 插到最前
        firstGpuRealCount++;
      }
    }
    /*
    这是整个函数最关键的一行，也是 ab2b89c（2.21.5-1，2024-03-26）新增的。原始版本这里是 *countPtr = count，只提前不截断。
    推断动机：MNNVL 把 ngpus 从 8 扩到最多 72，分支因子从 7 涨到 71，1<<19 的搜索预算根本不够。截断到 ≤2 让每步分支因子恒定。
    截断的结构性后果：结合 used 拦截，第一步选定方向后，回头的那个邻居必然已占用，于是整条环被锁死成 PCI 连续游走，排列空间从 (ngpus-1)! 塌缩到 2（升序与降序）。
    这个剪枝在全互连 fabric 上代价为零，因为所有 GPU 对带宽与跳数相同，排列之间本无质量差别，真正需要区分的是网卡分配，而那由 RecNet 负责。
    */
    *countPtr = firstGpuRealCount;
  }
  return ncclSuccess;
}
```

## Socket 明文握手状态机

新版由 crypt.cc 中的 ncclCryptAcceptHello / ncclCryptConnectHello 统一处理 hello，并支持 TLS；不再分别推进 magic 与 type 的旧状态机。

原文件：`src/misc/socket.cc`，以下为原有注释及邻近代码的摘录。

```cpp
  /*
  Linux 不继承的是 file status flags（O_NONBLOCK/O_ASYNC 这一族，存在 struct file 里，accept 会新建 file 结构）。
  而 TCP_NODELAY、SO_SNDBUF 这类 socket 选项，Linux 上 accept 出的新 fd 实际会从监听 socket 继承（BSD 派生实现的传统，Linux 亦然）。
  所以对 root 侧的阻塞 socket，这次重设严格说是幂等的防御性操作（跨平台统一、无害），并非功能必需。
  */
  // once accepted, linux sockets do NOT inherit file status flags such as O_NONBLOCK (BSD ones do)
  NCCLCHECK(ncclOsSocketSetFlags(sock));

  // 接收并校验 magic
  // 如果是同步，或者异步但是 magic 尚未收满（finalizeCounter < 8）
  if (sock->asyncFlag == 0 || sock->finalizeCounter < sizeof(magic)) {
    if (sock->asyncFlag == 0) {
      received = 0;
      // socketWait 忙等 or poll 读到指定 sizeof(magic) 字节大小的数据到 magic 中为止
      if (socketWait(NCCL_SOCKET_RECV, sock, &magic, sizeof(magic), &received) != ncclSuccess) {
        // 日志输出异常对端节点信息
        INFO(NCCL_NET | NCCL_INIT, "socketFinalizeAccept: handshake receive from %s failed, discarding peer connection",
             ncclSocketToString(&sock->addr, line));
        // 关闭 accepted 对端 socket，初始化状态
        ncclOsSocketResetAccept(sock);
        return ncclSuccess;
      }
    } else {
      // 异步
      int closed = 0;
      // 从上次的位置开始续读
      received = sock->finalizeCounter;
      // 推进一次接收操作
      NCCLCHECK(socketProgress(NCCL_SOCKET_RECV, sock, sock->finalizeBuffer, sizeof(magic), &received, &closed));
      sock->finalizeCounter = received;
      // 如果还没接收到预期数据大小
      if (received < sizeof(magic)) {
        if (closed) {  // 可能对端关闭了
          INFO(NCCL_NET | NCCL_INIT, "socketFinalizeAccept: peer %s closed before magic handshake, discarding",
               ncclSocketToString(&sock->addr, line));
          // 关闭 accepted 对端 socket，初始化状态
          ncclOsSocketResetAccept(sock);
        }
        return ncclSuccess;
      }
      // 本次将 sizeof(magic) 大小的数据异步读取完毕，拷贝到 magic 中。
      memcpy(&magic, sock->finalizeBuffer, sizeof(magic));
    }
    // 校验对比接收到的 magic 是否与 root 的 magic 一样
    if (magic != sock->magic) {
      INFO(NCCL_NET | NCCL_INIT,
           "socketFinalizeAccept from %s: socket magic mismatch (peer 0x%016llx != expected 0x%016llx), discarding "
           "peer connection",
           ncclSocketToString(&sock->addr, line), (unsigned long long)magic, (unsigned long long)sock->magic);
      // 关闭 accepted 对端 socket，初始化状态
      ncclOsSocketResetAccept(sock);
      return ncclSuccess;
    }
  }

  //接收 type
  if (sock->asyncFlag == 0) {
    // 同步直接读
    received = 0;
    NCCLCHECK(socketWait(NCCL_SOCKET_RECV, sock, &type, sizeof(type), &received));
  } else {
    // sock->finalizeCounter 包含 magic + type 的字节大小
    received = sock->finalizeCounter - sizeof(magic);
    // 异步读到临时 buffer
    NCCLCHECK(socketProgress(NCCL_SOCKET_RECV, sock, sock->finalizeBuffer, sizeof(type), &received));
    sock->finalizeCounter = received + sizeof(magic);
    // 异步没读完下次继续
    if (received < sizeof(type)) return ncclSuccess;
    // 读完了将临时 buffer 的数据拷贝到 type 中
    memcpy(&type, sock->finalizeBuffer, sizeof(type));
  }
  // 校验对比接收到的 type 是否与 root 的 type 一样
  if (type != sock->type) {
    INFO(NCCL_NET | NCCL_INIT,
         "socketFinalizeAccept from %s: wrong socket type (peer %d != expected %d -- peer connected to wrong NCCL "
         "socket), discarding peer connection",
         ncclSocketToString(&sock->addr, line), (int)type, (int)sock->type);
    ncclOsSocketResetAccept(sock);
    return ncclSuccess;
  } else {
    sock->state = ncclSocketStateReady;
  }
  return ncclSuccess;
}

ncclResult_t ncclSocketPollConnect(struct ncclSocket* sock) {
  if (sock == NULL) {
    WARN("ncclSocketPollConnect: pass NULL socket");
    return ncclInvalidArgument;
  }
  NCCLCHECK(ncclOsSocketPollConnect(sock));

static ncclResult_t socketFinalizeConnect(struct ncclSocket* sock) {
  int sent;
  if (sock->asyncFlag == 0) {
    // 同步循环依次发送 magic 和 type
    sent = 0;
    NCCLCHECK(socketWait(NCCL_SOCKET_SEND, sock, &sock->magic, sizeof(sock->magic), &sent));
    sent = 0;
    NCCLCHECK(socketWait(NCCL_SOCKET_SEND, sock, &sock->type, sizeof(sock->type), &sent));
  } else {
    // 异步循环依次发送 magic 和 type
    if (sock->finalizeCounter < sizeof(sock->magic)) {
      sent = sock->finalizeCounter;
      NCCLCHECK(socketProgress(NCCL_SOCKET_SEND, sock, &sock->magic, sizeof(sock->magic), &sent));
      sock->finalizeCounter = sent;
      // 本次没发送完，回到外层调用循环
      if (sent < sizeof(sock->magic)) return ncclSuccess;
    }
    sent = sock->finalizeCounter - sizeof(sock->magic);
    NCCLCHECK(socketProgress(NCCL_SOCKET_SEND, sock, &sock->type, sizeof(sock->type), &sent));
    sock->finalizeCounter = sent + sizeof(sock->magic);
    // 本次没发送完，回到外层调用
    if (sent < sizeof(sock->type)) return ncclSuccess;
  }
  sock->state = ncclSocketStateReady;
  return ncclSuccess;
}
```
